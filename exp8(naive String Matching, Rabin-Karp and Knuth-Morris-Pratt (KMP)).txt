#include <iostream>
#include <string>
#include <vector>
using namespace std;

//========== 8.1 NAIVE STRING MATCHING ==========

void naiveSearch(const string& text, const string& pattern) {
    int n = text.length();
    int m = pattern.length();

    if (m == 0 || m > n)
        return;

    for (int i = 0; i <= n - m; i++) {
        int j;

        for (j = 0; j < m; j++) {
            if (text[i + j] != pattern[j])
                break;
        }

        if (j == m)
            cout << "Pattern found at index: "
                 << i << endl;
    }
}

//========== 8.2 RABIN-KARP ALGORITHM ==========

void rabinKarp(const string& text, const string& pattern) {
    int n = text.length();
    int m = pattern.length();

    if (m == 0 || m > n)
        return;

    int d = 256;
    int q = 101;

    int p = 0;
    int t = 0;
    int h = 1;

    // Calculate d^(m-1) % q

    for (int i = 0; i < m - 1; i++)
        h = (h * d) % q;

    // Calculate initial hash values

    for (int i = 0; i < m; i++) {
        p = (d * p + (unsigned char)pattern[i]) % q;
        t = (d * t + (unsigned char)text[i]) % q;
    }

    // Slide pattern over text

    for (int i = 0; i <= n - m; i++) {
        if (p == t) {
            int j;

            for (j = 0; j < m; j++) {
                if (text[i + j] != pattern[j])
                    break;
            }

            if (j == m)
                cout << "Pattern found at index: "
                     << i << endl;
        }

        // Calculate hash for next window

        if (i < n - m) {
            t = (d * (t - (unsigned char)text[i] * h)
                 + (unsigned char)text[i + m]) % q;

            if (t < 0)
                t += q;
        }
    }
}

//========== 8.3 KMP ALGORITHM ==========

// Construct LPS array

void computeLPS(const string& pattern, vector<int>& lps) {
    int m = pattern.length();

    int len = 0;
    int i = 1;

    lps[0] = 0;

    while (i < m) {
        if (pattern[i] == pattern[len]) {
            len++;
            lps[i] = len;
            i++;
        }

        else {
            if (len != 0)
                len = lps[len - 1];

            else {
                lps[i] = 0;
                i++;
            }
        }
    }
}

// Perform KMP searching

void KMP(const string& text, const string& pattern) {
    int n = text.length();
    int m = pattern.length();

    if (m == 0 || m > n)
        return;

    vector<int> lps(m, 0);

    computeLPS(pattern, lps);

    int i = 0;
    int j = 0;

    while (i < n) {
        if (text[i] == pattern[j]) {
            i++;
            j++;
        }

        if (j == m) {
            cout << "Pattern found at index: "
                 << i - j << endl;

            j = lps[j - 1];
        }

        else if (i < n && text[i] != pattern[j]) {
            if (j != 0)
                j = lps[j - 1];

            else
                i++;
        }
    }
}

//========== MAIN FUNCTION ==========

int main() {
    string text = "AABAACAADAABAABA";
    string pattern = "AABA";

    cout << "Text: " << text << endl;
    cout << "Pattern: " << pattern << endl;

    // Naive String Matching

    cout << "\nNaive String Matching:\n";
    naiveSearch(text, pattern);

    // Rabin-Karp Algorithm

    cout << "\nRabin-Karp Algorithm:\n";
    rabinKarp(text, pattern);

    // KMP Algorithm

    cout << "\nKMP Algorithm:\n";
    KMP(text, pattern);

    return 0;
}
