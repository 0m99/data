#include <iostream>
using namespace std;

//========== 2.1 LINEAR SEARCH ==========

int linearSearch(int arr[], int n, int key) {
    for (int i = 0; i < n; i++) {
        if (arr[i] == key)
            return i;
    }

    return -1;
}

//========== 2.2 BINARY SEARCH ==========

int binarySearch(int arr[], int n, int key) {
    int low = 0;
    int high = n - 1;

    while (low <= high) {
        int mid = low + (high - low) / 2;

        if (arr[mid] == key)
            return mid;

        else if (arr[mid] < key)
            low = mid + 1;

        else
            high = mid - 1;
    }

    return -1;
}

//========== MAIN FUNCTION ==========

int main() {
    int arr[] = {10, 20, 30, 40, 50};
    int n = sizeof(arr) / sizeof(arr[0]);
    int key = 40;

    // Linear Search
    int result1 = linearSearch(arr, n, key);

    if (result1 == -1)
        cout << "Linear Search: Element not found\n";
    else
        cout << "Linear Search: Element found at index "
             << result1 << endl;

    // Binary Search
    int result2 = binarySearch(arr, n, key);

    if (result2 == -1)
        cout << "Binary Search: Element not found\n";
    else
        cout << "Binary Search: Element found at index "
             << result2 << endl;

    return 0;
}
