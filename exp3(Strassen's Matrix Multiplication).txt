#include <iostream>
#include <vector>
using namespace std;

typedef vector<vector<int>> Matrix;

//========== MATRIX ADDITION ==========

Matrix add(const Matrix& A, const Matrix& B, int n) {
    Matrix C(n, vector<int>(n));

    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            C[i][j] = A[i][j] + B[i][j];

    return C;
}

//========== MATRIX SUBTRACTION ==========

Matrix subtract(const Matrix& A, const Matrix& B, int n) {
    Matrix C(n, vector<int>(n));

    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            C[i][j] = A[i][j] - B[i][j];

    return C;
}

//========== STRASSEN'S ALGORITHM ==========

Matrix strassen(const Matrix& A, const Matrix& B, int n) {
    Matrix C(n, vector<int>(n));

    if (n == 1) {
        C[0][0] = A[0][0] * B[0][0];
        return C;
    }

    int k = n / 2;

    Matrix A11(k, vector<int>(k));
    Matrix A12 = A11, A21 = A11, A22 = A11;
    Matrix B11 = A11, B12 = A11;
    Matrix B21 = A11, B22 = A11;

    // Divide matrices into four submatrices

    for (int i = 0; i < k; i++) {
        for (int j = 0; j < k; j++) {
            A11[i][j] = A[i][j];
            A12[i][j] = A[i][j + k];
            A21[i][j] = A[i + k][j];
            A22[i][j] = A[i + k][j + k];

            B11[i][j] = B[i][j];
            B12[i][j] = B[i][j + k];
            B21[i][j] = B[i + k][j];
            B22[i][j] = B[i + k][j + k];
        }
    }

    // Calculate seven products

    Matrix M1 = strassen(add(A11, A22, k),
                         add(B11, B22, k), k);

    Matrix M2 = strassen(add(A21, A22, k),
                         B11, k);

    Matrix M3 = strassen(A11,
                         subtract(B12, B22, k), k);

    Matrix M4 = strassen(A22,
                         subtract(B21, B11, k), k);

    Matrix M5 = strassen(add(A11, A12, k),
                         B22, k);

    Matrix M6 = strassen(subtract(A21, A11, k),
                         add(B11, B12, k), k);

    Matrix M7 = strassen(subtract(A12, A22, k),
                         add(B21, B22, k), k);

    // Calculate resultant submatrices

    Matrix C11 = add(subtract(add(M1, M4, k),
                     M5, k), M7, k);

    Matrix C12 = add(M3, M5, k);

    Matrix C21 = add(M2, M4, k);

    Matrix C22 = add(add(subtract(M1, M2, k),
                     M3, k), M6, k);

    // Combine submatrices

    for (int i = 0; i < k; i++) {
        for (int j = 0; j < k; j++) {
            C[i][j] = C11[i][j];
            C[i][j + k] = C12[i][j];
            C[i + k][j] = C21[i][j];
            C[i + k][j + k] = C22[i][j];
        }
    }

    return C;
}

//========== MAIN FUNCTION ==========

int main() {
    Matrix A = {{1, 2}, {3, 4}};
    Matrix B = {{5, 6}, {7, 8}};

    int n = A.size();

    Matrix C = strassen(A, B, n);

    cout << "Resultant matrix:\n";

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++)
            cout << C[i][j] << " ";

        cout << endl;
    }

    return 0;
}
