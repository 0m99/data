#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
using namespace std;

//========== LONGEST COMMON SUBSEQUENCE ==========

void LCS(const string& X, const string& Y) {
    int m = X.length();
    int n = Y.length();

    vector<vector<int>> dp(m + 1,
                           vector<int>(n + 1, 0));

    // Construct dynamic programming table

    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (X[i - 1] == Y[j - 1])
                dp[i][j] = 1 + dp[i - 1][j - 1];
            else
                dp[i][j] = max(dp[i - 1][j],
                               dp[i][j - 1]);
        }
    }

    cout << "Length of LCS: " << dp[m][n] << endl;

    // Find the longest common subsequence

    string result = "";

    int i = m;
    int j = n;

    while (i > 0 && j > 0) {
        if (X[i - 1] == Y[j - 1]) {
            result += X[i - 1];
            i--;
            j--;
        }

        else if (dp[i - 1][j] > dp[i][j - 1])
            i--;

        else
            j--;
    }

    reverse(result.begin(), result.end());

    cout << "Longest Common Subsequence: " << result;
}

//========== MAIN FUNCTION ==========

int main() {
    string X = "AGGTAB";
    string Y = "GXTXAYB";

    LCS(X, Y);

    return 0;
}
